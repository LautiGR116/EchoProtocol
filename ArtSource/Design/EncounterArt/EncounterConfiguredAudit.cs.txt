using System;
using System.Linq;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.SceneManagement;
using EchoProtocol.Presentation;
using EchoProtocol.Campaign;
using EchoProtocol.Puzzle;
public static class EncounterConfiguredAudit {
 public static void Main(){var scene=SceneManager.GetActiveScene();if(UnityEditor.EditorApplication.isPlaying||scene.isDirty)throw new Exception("Saved edit scene required");
 var lines=new System.Collections.Generic.List<string>();lines.Add(scene.path+" saved; dirty=false");
 var notes=UnityEngine.Object.FindFirstObjectByType<ProtocolGuidance>();var ns=new SerializedObject(notes);foreach(var n in new[]{"firstRelay","firstEvent","domesticCheckpoint","alignmentRelay","passage","returnPassage","recoveryOrigin","returnRelay","campaign","house","finalReview","hubReturn"}){var o=ns.FindProperty(n).objectReferenceValue;if(!o)throw new Exception("Missing "+n);lines.Add("guidance "+n+" = "+o.name);}
 var feedback=UnityEngine.Object.FindFirstObjectByType<OrientationFeedback>();var fs=new SerializedObject(feedback);foreach(var n in new[]{"relay","readerA","readerB","statusA","statusB","statusC","review","interiorStatus","exteriorStatus","receivedDoor","doorStatus"}){var o=fs.FindProperty(n).objectReferenceValue;if(!o)throw new Exception("Missing "+n);lines.Add("feedback "+n+" = "+o.name);}
 foreach(var linked in UnityEngine.Object.FindObjectsByType<LinkedTaskLight>(FindObjectsSortMode.None)){var s=new SerializedObject(linked);var circuit=(Light)s.FindProperty("circuitSource").objectReferenceValue;var light=(Light)s.FindProperty("taskLight").objectReferenceValue;if(!circuit||!light)throw new Exception("Broken circuit");lines.Add("task "+linked.name+" <- "+circuit.name+" nominal="+s.FindProperty("sourceNominalIntensity").floatValue+" gain="+s.FindProperty("taskIntensity").floatValue);}
 var director=UnityEngine.Object.FindFirstObjectByType<CampaignDirector>();lines.Add("save="+new SerializedObject(director).FindProperty("checkpointKey").stringValue);
 lines.Add("lights="+UnityEngine.Object.FindObjectsByType<Light>(FindObjectsInactive.Include,FindObjectsSortMode.None).Length+" audio="+UnityEngine.Object.FindObjectsByType<AudioSource>(FindObjectsInactive.Include,FindObjectsSortMode.None).Length);
 int missing=scene.GetRootGameObjects().Sum(g=>g.GetComponentsInChildren<Transform>(true).Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)));if(missing!=0)throw new Exception("Missing scripts");lines.Add("missingScripts=0");
 var scenes=EditorBuildSettings.scenes.Where(s=>s.enabled).ToArray();if(scenes.Length!=1||scenes[0].path!=scene.path)throw new Exception("Build scenes mismatch");lines.Add("sole enabled build scene matches");
 var treatment=UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Audio.EncounterAtmosphere>(); if(!treatment)throw new Exception("Missing treatment");
 var ts=new SerializedObject(treatment);
 foreach(var n in new[]{"session","listener","passage","review","passageRegion","finalRegion","passageVent","finalVent","frameContact","passageContact","passageRotor","finalRotor"}){var o=ts.FindProperty(n).objectReferenceValue;if(!o)throw new Exception("Missing treatment "+n);lines.Add("atmosphere "+n+" = "+o.name);}
 foreach(var n in new[]{"passageVent","finalVent","frameContact"}){var a=(AudioSource)ts.FindProperty(n).objectReferenceValue;if(a.spatialBlend<.999f||a.dopplerLevel!=0||!a.outputAudioMixerGroup)throw new Exception("Unbound fixed source");var v=a.transform.position;var f=a.GetComponent<AudioLowPassFilter>();lines.Add(n+" pos="+v.ToString("F3")+" blend="+a.spatialBlend+" min="+a.minDistance+" max="+a.maxDistance+" pitch="+a.pitch+" loop="+a.loop+" group="+a.outputAudioMixerGroup.name+" lowpass="+(f?f.cutoffFrequency:0));}
 lines.Add("passageGain="+ts.FindProperty("passageVentGain").floatValue+" finalGain="+ts.FindProperty("finalVentGain").floatValue);
 foreach(var n in new[]{"passageRegion","finalRegion"}){var c=(BoxCollider)ts.FindProperty(n).objectReferenceValue;if(!c.isTrigger||c.GetComponents<MonoBehaviour>().Length!=0)throw new Exception("Region must have no game callbacks");lines.Add(n+" acoustic-only trigger");}
 foreach(var o in new UnityEngine.Object[]{UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Horror.VS1StrongEventOne>(),UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Audio.CampaignRoomTone>()}){if(new SerializedObject(o).FindProperty("encounterAtmosphere").objectReferenceValue!=treatment)throw new Exception("Gain owner not bound");lines.Add(o.name+" owns its gain; treatment assigned");}
 var loop=UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Echo.LoopController>();var prefab=(EchoProtocol.Echo.EchoPlayback)new SerializedObject(loop).FindProperty("echoPrefab").objectReferenceValue;var body=prefab.GetComponentsInChildren<MeshFilter>(true).Single(m=>m.name=="EP_AdultRecordedSurface");var depth=body.transform.Find("Recorded self depth").GetComponent<MeshFilter>();if(body.sharedMesh!=depth.sharedMesh||prefab.GetComponentsInChildren<AudioSource>(true).Length!=0||prefab.GetComponentsInChildren<Animator>(true).Length!=0)throw new Exception("Display contract failed");
 var mats=body.GetComponent<Renderer>().sharedMaterials;if(mats.Length!=8||mats.Any(m=>!m||m.shader.name!="EchoProtocol/RecordedWitness"))throw new Exception("Wrong surface palette");var shader=mats[0].shader;var messages=ShaderUtil.GetShaderMessages(shader);if(messages.Any(m=>m.severity.ToString()=="Error"))throw new Exception("Shader compile error");lines.Add("witness prefab="+AssetDatabase.GetAssetPath(prefab)+" shared self-depth mesh="+AssetDatabase.GetAssetPath(body.sharedMesh)+" vertices="+body.sharedMesh.vertexCount+" slots="+mats.Length+" AudioSource=0 Animator=0 shaderSupported="+shader.isSupported+" shaderMessages="+messages.Length);
 var review=UnityEngine.Object.FindFirstObjectByType<FinalDoorMemory>();var material=new SerializedObject(review).FindProperty("directWitnessMaterial").objectReferenceValue;if(material!=mats.Single(m=>m.name=="M_Record_FIG_Skin"))throw new Exception("Final direct witness is not skin slot");lines.Add("direct witness skin slot="+material.name+"; pass RecordedPresenceForward="+body.GetComponent<Renderer>().sharedMaterials[0].FindPass("RecordedPresenceForward"));
 var figure=UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None).Single(t=>t.name=="SE2 physical adult / same authored silhouette");if(figure.GetComponent<MeshFilter>().sharedMesh!=body.sharedMesh)throw new Exception("Figure identity mismatch");lines.Add("SE2 same body yaw="+figure.parent.eulerAngles.y+" shadow casting="+figure.GetComponent<Renderer>().shadowCastingMode);
 if(new SerializedObject(director).FindProperty("checkpointKey").stringValue!="EP.CampaignCheckpoint.EncounterArt.v1")throw new Exception("Wrong save key");
 File.WriteAllLines("Docs/VisualEvidence/2026-10-02-EncounterArt/configuration.txt",lines);
 }
}
