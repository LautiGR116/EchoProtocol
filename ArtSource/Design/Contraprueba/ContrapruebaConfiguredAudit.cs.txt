using System;using System.Linq;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEditor;using EchoProtocol.Echo;using EchoProtocol.Puzzle;using EchoProtocol.Horror;using EchoProtocol.Campaign;using Object=UnityEngine.Object;
public static class ContrapruebaConfiguredAudit{
 static T[] All<T>()where T:Component=>Object.FindObjectsByType<T>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(x=>x.gameObject.scene.IsValid()).ToArray();
 static T Ref<T>(Object o,string n)where T:Object=>(T)new SerializedObject(o).FindProperty(n).objectReferenceValue;
 public static object Main(){var checks=new List<string>();Action<bool,string> check=(ok,s)=>{if(!ok)throw new Exception(s);checks.Add(s);};check(!EditorApplication.isPlaying,"saved-scene audit in Edit mode");var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();check(scene.path=="Assets/_Game/Scenes/Production/EchoProtocol_Contraprueba.unity"&&!scene.isDirty,"correct independent scene saved");check(All<Transform>().Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject))==0,"no missing scripts");var relay=All<RecombinationRelay>().Single();var loop=Ref<LoopController>(relay,"loop");var recorder=Ref<EchoRecorder>(loop,"recorder");check(relay.RequiresPairedTransfer&&relay.RequiresReturnWitness,"three-stage reception opt-in");check(Ref<PlayerBodyPoseSource>(recorder,"bodyPoseSource"),"actual recorder has explicit captured body source");foreach(string name in new[]{"routeGate","finalGate"}){var d=Ref<SlidingDoor>(relay,name);var b=Ref<Rigidbody>(d,"doorBody");check(b&&b.isKinematic&&b.GetComponentsInChildren<Renderer>(true).All(r=>r.enabled)&&b.GetComponentsInChildren<Collider>(true).Any(c=>c.enabled&&!c.isTrigger),name+" explicit body is visible and solid");}
 var prefab=Ref<EchoPlayback>(loop,"echoPrefab");var rig=prefab.GetComponent<EchoBodyRig>();check(rig,"new explicit rig prefab assigned");foreach(var name in new[]{"modelRoot","hips","spine","head","leftThigh","leftShin","leftFoot","rightThigh","rightShin","rightFoot","leftArm","leftForearm","rightArm","rightForearm"})check(Ref<Transform>(rig,name),"rig reference "+name);var skins=prefab.GetComponentsInChildren<SkinnedMeshRenderer>(true);check(skins.Length==2&&skins.All(s=>s.bones.Length==16&&s.sharedMesh&&s.sharedMaterials.All(m=>m&&m.shader.isSupported)),"two matched supported skinned display surfaces with 16 bones");check(prefab.GetComponentsInChildren<AudioSource>(true).Length==0&&prefab.GetComponentsInChildren<Animator>(true).Length==0,"Echo adds no independent Animator or unrecorded sound");
 var atmosphere=All<ContrapruebaAtmosphere>().Single();check(Ref<RecombinationRelay>(atmosphere,"relay")==relay,"physical atmosphere follows actual relay");var reset=new SerializedObject(loop).FindProperty("resetTargets");check(reset!=null&&Enumerable.Range(0,reset.arraySize).Any(i=>reset.GetArrayElementAtIndex(i).objectReferenceValue==atmosphere),"RECALL explicitly resets physical atmosphere");var returnControl=All<ReturnWitnessControl>().Single();check(Ref<RecombinationRelay>(returnControl,"relay")==relay,"held return control shares relay");
 var campaign=All<CampaignDirector>().Single();check(new SerializedObject(campaign).FindProperty("checkpointKey").stringValue=="EP.CampaignCheckpoint.Contraprueba.v1","saves isolated from preserved editions");check(EditorBuildSettings.scenes.Count(s=>s.enabled)==1&&EditorBuildSettings.scenes.Single(s=>s.enabled).path==scene.path,"only this selected scene builds");var result=new{status="PASS",checks,scope="Saved serialized configuration, supported shaders and physical gate bodies. Behavioral evidence is separate."};File.WriteAllText("Docs/VisualEvidence/2026-10-08-Contraprueba/configured-audit.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));return result;}
}
