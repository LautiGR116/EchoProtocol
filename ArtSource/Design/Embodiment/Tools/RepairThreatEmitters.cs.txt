using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using EchoProtocol.Campaign;using EchoProtocol.Echo;
public static class RepairThreatEmitters {
 static T In<T>(UnityEngine.SceneManagement.Scene scene) where T:Component=>scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<T>(true)).Single();
 public static object Main(){if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(!scene.path.EndsWith("Embodiment.unity"))throw new Exception("Candidate required");var target=In<MachineCoverThreat>(scene);var old=EditorSceneManager.OpenScene("Assets/_Game/Scenes/Production/EchoProtocol_Coverage.unity",OpenSceneMode.Additive);
  try{var source=In<MachineCoverThreat>(old);var report=new List<object>();foreach(string field in new[]{"cry","movement","strike"}){var s=(AudioSource)new SerializedObject(source).FindProperty(field).objectReferenceValue;var so=new SerializedObject(target);if(so.FindProperty(field).objectReferenceValue)throw new Exception("Preserve assigned "+field);if(!s)throw new Exception("Original source missing "+field);var g=new GameObject(s.name);g.transform.SetParent(target.transform,false);g.transform.localPosition=source.transform.InverseTransformPoint(s.transform.position);g.transform.localRotation=Quaternion.Inverse(source.transform.rotation)*s.transform.rotation;var a=g.AddComponent<AudioSource>();EditorUtility.CopySerialized(s,a);so.FindProperty(field).objectReferenceValue=a;so.ApplyModifiedPropertiesWithoutUndo();report.Add(new{field,clip=AssetDatabase.GetAssetPath(a.clip),a.volume,a.spatialBlend,local=g.transform.localPosition.ToString("F3"),sourceComponents=s.GetComponents<Component>().Select(c=>c.GetType().Name).ToArray()});}
   EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);File.WriteAllText("Docs/VisualEvidence/2026-10-09-Embodiment/threat-emitter-repair.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{status="RESTORED",report,scope="Copies original Coverage emitters, clips, mixer groups and safe source settings. No new vocal asset and no puzzle/chase-rule change."},Newtonsoft.Json.Formatting.Indented));return report;
  }finally{EditorSceneManager.CloseScene(old,true);UnityEngine.SceneManagement.SceneManager.SetActiveScene(scene);}
 }
}
