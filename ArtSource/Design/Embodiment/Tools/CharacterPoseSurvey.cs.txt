using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;
using EchoProtocol.Echo;using EchoProtocol.Player;
public static class CharacterPoseSurvey {
 const string Out="Docs/VisualEvidence/2026-10-09-Embodiment/pose-survey";
 static void Shot(Camera c,string name,Vector3 pos,Vector3 aim){var old=c.targetTexture;var active=RenderTexture.active;var rt=new RenderTexture(1280,720,24);var t=new Texture2D(1280,720,TextureFormat.RGB24,false);try{c.transform.SetPositionAndRotation(pos,Quaternion.LookRotation(aim-pos));c.targetTexture=rt;c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,1280,720),0,0);t.Apply();File.WriteAllBytes(Out+"/"+name+".png",t.EncodeToPNG());}finally{c.targetTexture=old;RenderTexture.active=active;UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(t);}}
 public static object Main(){if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");Directory.CreateDirectory(Out);var c=UnityEngine.Object.FindFirstObjectByType<FirstPersonController>().GetComponentInChildren<Camera>();var cp=c.transform.position;var cr=c.transform.rotation;var list=new List<object>();
  var echo=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/_Game/Prefabs/Embodiment/EchoEmbodiment.prefab"));var threat=UnityEngine.Object.Instantiate(UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Campaign.MachineCoverThreat>().GetComponentInChildren<RecordedCharacterMotion>().gameObject);var owned=new[]{echo,threat};foreach(var g in owned)foreach(var r in g.GetComponentsInChildren<SkinnedMeshRenderer>())r.forceMatrixRecalculationPerRender=true;
  try{foreach(var choice in new[]{new{g=echo,name="Echo"},new{g=threat,name="Pursuer"}}){var g=choice.g;foreach(var other in owned)other.SetActive(other==g);g.transform.SetPositionAndRotation(new Vector3(0,.03f,-8),Quaternion.Euler(0,180,0));var r=g.GetComponent<EchoBodyRig>();var m=g.GetComponent<RecordedCharacterMotion>();m.Initialize();m.SetPresentIntent(choice.name=="Pursuer"?1.6f:-1);
    foreach(var pose in new[]{new{label="idle",speed=Vector2.zero,phase=0f,turn=0f,turnPhase=0f,grip=false},new{label="walk",speed=new Vector2(0,1.55f),phase=1.8f,turn=0f,turnPhase=0f,grip=false},new{label="run",speed=new Vector2(0,3.5f),phase=1.8f,turn=0f,turnPhase=0f,grip=false},new{label="strafe",speed=new Vector2(3.5f,0),phase=2.8f,turn=0f,turnPhase=0f,grip=false},new{label="back",speed=new Vector2(0,-3.5f),phase=2.8f,turn=0f,turnPhase=0f,grip=false},new{label="turn",speed=Vector2.zero,phase=0f,turn=120f,turnPhase=1.8f,grip=false},new{label="grip",speed=Vector2.zero,phase=0f,turn=0f,turnPhase=0f,grip=true}}){r.Apply(new RecordedBodyPose(pose.phase,1.1f,0,pose.speed,pose.turnPhase,pose.turn));r.ApplyMachineGrip(pose.grip);Shot(c,choice.name+"-"+pose.label,new Vector3(0,1.6f,-11.3f),new Vector3(0,1.0f,-8));if(pose.label=="idle"||pose.label=="grip")Shot(c,choice.name+"-"+pose.label+"-near",new Vector3(.2f,1.66f,-8.95f),new Vector3(0,1.68f,-8));
     var sm=g.GetComponentInChildren<SkinnedMeshRenderer>();var b=new Mesh();sm.BakeMesh(b);var vs=b.vertices;float min=vs.Min(v=>sm.transform.TransformPoint(v).y);list.Add(new{choice.name,pose.label,meshMinY=min,left=m.LeftFoot.position.ToString("F3"),right=m.RightFoot.position.ToString("F3"),finite=vs.All(v=>float.IsFinite(v.x)&&float.IsFinite(v.y)&&float.IsFinite(v.z))});UnityEngine.Object.DestroyImmediate(b);}
   }var report=new{status="SURVEYED",scope="Editor visual/deformation fixtures with actual Player camera and authored scene light. No live recording claimed.",poses=list};File.WriteAllText(Out+"/survey.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));return report;
  }finally{c.transform.SetPositionAndRotation(cp,cr);foreach(var g in owned)UnityEngine.Object.DestroyImmediate(g);}
 }
}
