using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using EchoProtocol.Campaign;
public static class CloneEmbodiment{public static string Main(){
const string src="Assets/_Game/Scenes/Production/EchoProtocol_Coverage.unity", dst="Assets/_Game/Scenes/Production/EchoProtocol_Embodiment.unity";
if(EditorApplication.isPlaying||UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty)throw new System.Exception("Clean stopped Editor required");
if(AssetDatabase.LoadAssetAtPath<SceneAsset>(dst))throw new System.Exception("Candidate already exists");
if(!AssetDatabase.CopyAsset(src,dst))throw new System.Exception("Copy failed");var scene=EditorSceneManager.OpenScene(dst);
var so=new SerializedObject(Object.FindFirstObjectByType<CampaignDirector>());so.FindProperty("checkpointKey").stringValue="EP.CampaignCheckpoint.Embodiment.v1";so.ApplyModifiedPropertiesWithoutUndo();
EditorSceneManager.SaveScene(scene);return scene.path;}}
