using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;
public static class AuthorCharacterClips {
 const string B="Assets/_Game/Art/Embodiment/RenderpeoplePrivate/";
 static string C(string n)=>n.Replace("rp_nathan_animated_003_walking_","");
 static void Save(UnityEngine.Object x,string p){if(AssetDatabase.LoadAssetAtPath<UnityEngine.Object>(p))throw new Exception("Preserve existing asset "+p);AssetDatabase.CreateAsset(x,p);}
 static void Rotate(Transform t,Vector3 axis,float angle){t.rotation=Quaternion.AngleAxis(angle,axis)*t.rotation;}
 static void Curves(AnimationClip clip,Transform root,Transform[] bones,List<Quaternion[]> rs,List<Vector3[]> ps,float dt){
  for(int j=0;j<bones.Length;j++){string path=AnimationUtility.CalculateTransformPath(bones[j],root);for(int a=0;a<4;a++){var c=new AnimationCurve();for(int f=0;f<rs.Count;f++)c.AddKey(f*dt,rs[f][j][a]);for(int f=0;f<c.length;f++){AnimationUtility.SetKeyLeftTangentMode(c,f,AnimationUtility.TangentMode.Linear);AnimationUtility.SetKeyRightTangentMode(c,f,AnimationUtility.TangentMode.Linear);}AnimationUtility.SetEditorCurve(clip,EditorCurveBinding.FloatCurve(path,typeof(Transform),"m_LocalRotation."+"xyzw"[a]),c);}
   for(int a=0;a<3;a++){var c=new AnimationCurve();for(int f=0;f<ps.Count;f++)c.AddKey(f*dt,ps[f][j][a]);for(int f=0;f<c.length;f++){AnimationUtility.SetKeyLeftTangentMode(c,f,AnimationUtility.TangentMode.Linear);AnimationUtility.SetKeyRightTangentMode(c,f,AnimationUtility.TangentMode.Linear);}AnimationUtility.SetEditorCurve(clip,EditorCurveBinding.FloatCurve(path,typeof(Transform),"m_LocalPosition."+"xyz"[a]),c);}}
  clip.EnsureQuaternionContinuity();
 }

   string model=who=="Eric"?"rp_eric_rigged_001_u3d.fbx":"rp_nathan_animated_003_walking.fbx";var g=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(B+who+"/"+model));
   try{foreach(var a in g.GetComponentsInChildren<Animator>())UnityEngine.Object.DestroyImmediate(a);var bones=g.GetComponentsInChildren<Transform>().Where(t=>t!=g.transform&&!t.GetComponent<SkinnedMeshRenderer>()).ToArray();var d=bones.ToDictionary(t=>C(t.name));var bindR=bones.Select(t=>t.localRotation).ToArray();var bindP=bones.Select(t=>t.localPosition).ToArray();var source=AssetDatabase.LoadAssetAtPath<AnimationClip>(B+who+"/RecordedWalk.anim");
    source.SampleAnimation(g,25/30f);var fingerR=bones.Select(t=>t.localRotation).ToArray();
    for(int i=0;i<bones.Length;i++){bones[i].localRotation=bindR[i];bones[i].localPosition=bindP[i];if(new[]{"thumb","index","middle","ring","pinky"}.Any(k=>C(bones[i].name).StartsWith(k)))bones[i].localRotation=fingerR[i];}
    Rotate(d["upperarm_l"],g.transform.forward,81);Rotate(d["upperarm_r"],g.transform.forward,-81);Rotate(d["lowerarm_l"],g.transform.right,-7);Rotate(d["lowerarm_r"],g.transform.right,-7);Rotate(d["hand_l"],g.transform.up,-6);Rotate(d["hand_r"],g.transform.up,6);
    var neutralR=bones.Select(t=>t.localRotation).ToArray();var neutralP=bones.Select(t=>t.localPosition).ToArray();var idle=new AnimationClip{name=who+" / authored quiet stance",legacy=true,frameRate=30,wrapMode=WrapMode.ClampForever};Curves(idle,g.transform,bones,new List<Quaternion[]>{neutralR,neutralR},new List<Vector3[]>{neutralP,neutralP},1/30f);Save(idle,B+who+"/QuietStance.anim");
    var rr=new List<Quaternion[]>();var pp=new List<Vector3[]>();for(int f=0;f<=34;f++){for(int j=0;j<bones.Length;j++){bones[j].localPosition=bindP[j];bones[j].localRotation=bindR[j];}source.SampleAnimation(g,(25+f)/30f);d["root"].localPosition=Vector3.zero;d["root"].localRotation=Quaternion.identity;var r=bones.Select(t=>t.localRotation).ToArray();var p=bones.Select(t=>t.localPosition).ToArray();if(f>=31){float blend=Mathf.SmoothStep(0,1,(f-30)/4f);for(int j=0;j<bones.Length;j++){r[j]=Quaternion.Slerp(r[j],rr[0][j],blend);p[j]=Vector3.Lerp(p[j],pp[0][j],blend);}}if(f>0)for(int j=0;j<bones.Length;j++)if(Quaternion.Dot(rr[f-1][j],r[j])<0)r[j]=new Quaternion(-r[j].x,-r[j].y,-r[j].z,-r[j].w);rr.Add(r);pp.Add(p);}
    var walk=new AnimationClip{name=who+" / continuous captured-phase cycle",legacy=true,frameRate=30,wrapMode=WrapMode.Loop};Curves(walk,g.transform,bones,rr,pp,1/30f);Save(walk,B+who+"/WalkingCycle.anim");

No clips run autonomously."};File.WriteAllText("/Users/lautigr116/EchoProtocol-licensed/Renderpeople-20261009/character-clips.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));return report;
 }
