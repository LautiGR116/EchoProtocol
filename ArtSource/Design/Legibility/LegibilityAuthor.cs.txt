 static T Get<T>(Object o,string n) where T:Object => (T)new SerializedObject(o).FindProperty(n).objectReferenceValue;
 static GameObject Box(string n,Transform parent,Vector3 p,Vector3 size,Quaternion q){var g=GameObject.CreatePrimitive(PrimitiveType.Cube);g.name=n;g.transform.SetParent(parent);g.transform.SetPositionAndRotation(p,q);g.transform.localScale=size;g.GetComponent<Renderer>().sharedMaterial=steel;Object.DestroyImmediate(g.GetComponent<Collider>());return g;}
 static TextMesh Plate(string n,string text,Vector3 p,float yaw,float width,float height,bool display=false,Transform parent=null)
 {
  parent=parent?parent:root;var q=Quaternion.Euler(0,yaw,0);
  Box(n+" / enamel plate",parent,p+q*Vector3.forward*.018f,new Vector3(width+.08f,height+.06f,.035f),q);
  var t=Object.Instantiate(template,parent);t.name=n;t.gameObject.SetActive(true);t.transform.SetPositionAndRotation(p,q);t.transform.localScale=Vector3.one;
  t.text=text;t.fontSize=72;t.characterSize=.01f;t.anchor=TextAnchor.MiddleCenter;t.alignment=TextAlignment.Center;t.color=display?new Color(.57f,.77f,.69f):new Color(.87f,.85f,.73f);
  var shader=Shader.Find(display?"EchoProtocol/DiegeticText":"EchoProtocol/PrintedText"); if(!shader)throw new Exception("Missing text shader");
  Ref(t.GetComponent<DiegeticText>(),"depthTestedShader",shader);
  float w=t.GetComponent<Renderer>().localBounds.size.x;if(w>width)t.characterSize*=width/w;
  return t;
 }
 static void Fit(string n,string words,float width){var t=Named(n).GetComponent<TextMesh>();t.text=words;t.characterSize=.014f;float w=t.GetComponent<Renderer>().localBounds.size.x;if(w>width)t.characterSize*=width/w;}

 {

  var exact=All<MonoBehaviour>().Where(m=>m is OpticalReader||m is ParallelObservationRelay||m is RecombinationRelay||m is BlindAlignmentRelay||m is FinalDoorMemory||m is EchoProtocol.Echo.LoopController||m is EchoProtocol.Echo.EchoRecorder).Cast<Object>().Concat(All<Collider>()).Concat(All<AudioSource>()).ToDictionary(o=>o,EditorJsonUtility.ToJson);
