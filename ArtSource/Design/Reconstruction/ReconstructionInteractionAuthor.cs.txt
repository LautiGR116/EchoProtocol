public static class ReconstructionInteractionAuthor {
 static void Ref(Object o,string n,Object v){var s=new SerializedObject(o);s.FindProperty(n).objectReferenceValue=v;s.ApplyModifiedPropertiesWithoutUndo();}
 public static object Main(){var scene=SceneManager.GetActiveScene();if(EditorApplication.isPlaying||scene.path!="Assets/_Game/Scenes/Production/EchoProtocol_Reconstruction.unity"||scene.isDirty)throw new Exception("Saved Reconstruction Edit required");if(Object.FindFirstObjectByType<InteractionFocusHUD>())throw new Exception("Already authored");
 var target=Object.FindObjectsByType<HouseInteraction>(FindObjectsSortMode.None).Single(h=>h.Kind==HouseInteractionKind.Telephone);var renders=Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None);var body=renders.Single(r=>r.name=="TelephoneBody");var handset=renders.Single(r=>r.name=="TelephoneHandset");
 var box=target.GetComponent<BoxCollider>();var bounds=body.bounds;var local=new Bounds(target.transform.InverseTransformPoint(bounds.center),Vector3.zero);for(int x=-1;x<=1;x+=2)for(int y=-1;y<=1;y+=2)for(int z=-1;z<=1;z+=2)local.Encapsulate(target.transform.InverseTransformPoint(bounds.center+Vector3.Scale(bounds.extents,new Vector3(x,y,z))));box.center=local.center;box.size=local.size;
 var handBox=handset.gameObject.AddComponent<BoxCollider>();var mesh=handset.GetComponent<MeshFilter>().sharedMesh;handBox.center=mesh.bounds.center;handBox.size=mesh.bounds.size;var surface=handset.gameObject.AddComponent<InteractionSurface>();Ref(surface,"owner",target);
 var interactor=Object.FindFirstObjectByType<PlayerInteractor>();var input=new SerializedObject(interactor);input.FindProperty("focusRadius").floatValue=.075f;input.FindProperty("focusAngle").floatValue=1.6f;input.FindProperty("releaseMargin").floatValue=.08f;input.ApplyModifiedPropertiesWithoutUndo();
 var session=Object.FindFirstObjectByType<GameSession>();var hud=new GameObject("Interaction / stable focus and actual E action").AddComponent<InteractionFocusHUD>();Ref(hud,"session",session);Ref(hud,"interactor",interactor);Ref(hud,"view",interactor.GetComponentInChildren<Camera>());Ref(hud,"campaign",Object.FindFirstObjectByType<CampaignDirector>());Ref(hud,"isolation",Object.FindFirstObjectByType<PressureIsolation>());Ref(session,"interactionFocusHud",hud);
 Physics.SyncTransforms();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);AssetDatabase.SaveAssets();File.WriteAllText("Docs/VisualEvidence/2026-10-02-Reconstruction/interaction-authoring.txt","Editor API: telephone owner collider fitted to visible static body. Handset mesh-local collider follows actual articulated receiver with explicit same-owner InteractionSurface. Stable focus HUD has explicit session/interactor/camera/campaign/hold references. Focus radius .075m with1.6deg angular bound and8cm release margin; direct physical sightline mandatory. Scene saved.\n");return "Interaction surfaces and focus HUD authored";
 }
}
