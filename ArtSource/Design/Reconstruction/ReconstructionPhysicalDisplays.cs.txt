using System;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using UnityEngine.SceneManagement;using Object=UnityEngine.Object;
public static class ReconstructionPhysicalDisplays {
static Transform N(string n)=>Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None).Single(t=>t.name==n);
public static object Main(){if(EditorApplication.isPlaying)throw new Exception("Edit only");var screen=AssetDatabase.LoadAssetAtPath<Material>("Assets/_Game/Art/Reconstruction/M_ReceiverHousing.mat");var root=new GameObject("Reconstruction / physically mounted P1 labels").transform;
foreach(string id in new[]{"A","B","C"}){var t=N("P1 "+id+" actual state");var pos=t.position;pos.y=1.73f;t.position=pos;var plate=N("P1 "+id+" actual state / enamel plate");plate.position=pos+Vector3.right*.022f;plate.localScale=new Vector3(.035f,.215f,1.14f);plate.GetComponent<Renderer>().sharedMaterial=screen;}
foreach(var name in new[]{"P1 A rule","P1 B rule","P1 service rule"}){var text=N(name);var plate=GameObject.CreatePrimitive(PrimitiveType.Cube);plate.name=name+" / physical instruction plaque";plate.transform.SetParent(root,false);plate.transform.position=text.position+Vector3.right*.026f;plate.transform.localScale=new Vector3(.038f,.24f,1.15f);plate.GetComponent<Renderer>().sharedMaterial=screen;Object.DestroyImmediate(plate.GetComponent<Collider>());}
var scene=SceneManager.GetActiveScene();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);AssetDatabase.SaveAssets();return "P1 labels/display backgrounds moved together; status now above optical case, no floating text over hardware";}}
