using System;using System.IO;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using UnityEngine.Rendering;using UnityEngine.SceneManagement;using EchoProtocol.Echo;using EchoProtocol.Campaign;using EchoProtocol.Horror;using EchoProtocol.Puzzle;using Object=UnityEngine.Object;
public static class ReconstructionModelAuthor {
const string Art="Assets/_Game/Art/Reconstruction/";
static Object Ref(Object o,string n)=>new SerializedObject(o).FindProperty(n).objectReferenceValue;
static void Set(Object o,string n,Object v){var s=new SerializedObject(o);s.FindProperty(n).objectReferenceValue=v;s.ApplyModifiedPropertiesWithoutUndo();}
static Color ColorFor(string n){switch(n){case "FIG_Skin":return new Color(.50f,.40f,.32f);case "FIG_Hair":return new Color(.043f,.039f,.034f);case "FIG_Eye":return new Color(.33f,.32f,.29f);case "FIG_Iris":return new Color(.073f,.081f,.075f);case "FIG_Trousers":return new Color(.13f,.145f,.15f);case "FIG_Shoe":return new Color(.09f,.079f,.067f);case "FIG_Sole":return new Color(.04f,.036f,.03f);case "FIG_Stitch":return new Color(.24f,.25f,.21f);case "FIG_Metal":return new Color(.22f,.23f,.20f);default:return new Color(.24f,.29f,.27f);}}
static Material Mat(string n,bool record){string path=Art+(record?"M_Record_":"M_Physical_")+n+".mat";var m=AssetDatabase.LoadAssetAtPath<Material>(path);if(!m){m=new Material(Shader.Find(record?"EchoProtocol/RecordedWitness":"Universal Render Pipeline/Lit"));m.name=(record?"M_Record_":"M_Physical_")+n;AssetDatabase.CreateAsset(m,path);}m.SetColor(record?"_Color":"_BaseColor",ColorFor(n));if(record){m.SetColor("_RecordTint",new Color(.08f,.30f,.28f));m.SetFloat("_BodyOpacity",.96f);m.SetFloat("_EdgeOpacity",.02f);m.SetFloat("_WorldLightGain",.90f);m.SetFloat("_RecordLevel",.28f);m.SetFloat("_NearFadeStart",.35f);m.SetFloat("_NearFadeEnd",.95f);m.SetFloat("_Weave",n=="FIG_Jacket"||n=="FIG_Trousers"?.07f:0);m.renderQueue=3000;}else{m.SetFloat("_Smoothness",n=="FIG_Eye"?.3f:.16f);m.SetFloat("_Metallic",0);}EditorUtility.SetDirty(m);return m;}
public static object Main(){var scene=SceneManager.GetActiveScene();if(EditorApplication.isPlaying||scene.path!="Assets/_Game/Scenes/Production/EchoProtocol_Reconstruction.unity"||scene.isDirty)throw new Exception("Saved Reconstruction Edit required");AssetDatabase.Refresh();string path="Assets/_Game/Art/Models/Reconstruction/EP_ReconstructedAdult.fbx";var imp=(ModelImporter)AssetImporter.GetAtPath(path);var prior=(ModelImporter)AssetImporter.GetAtPath("Assets/_Game/Art/Models/EncounterArt/EP_WitnessSurface.fbx");imp.globalScale=prior.globalScale;imp.useFileScale=prior.useFileScale;imp.bakeAxisConversion=prior.bakeAxisConversion;imp.importAnimation=false;imp.isReadable=true;imp.importNormals=ModelImporterNormals.Import;imp.SaveAndReimport();var model=AssetDatabase.LoadAssetAtPath<GameObject>(path);var mesh=model.GetComponentInChildren<MeshFilter>().sharedMesh;var slots=model.GetComponentInChildren<Renderer>().sharedMaterials.Select(m=>m.name).ToArray();var recorded=slots.Select(n=>Mat(n,true)).ToArray();var physical=slots.Select(n=>Mat(n,false)).ToArray();
var loop=Object.FindFirstObjectByType<LoopController>();var original=(EchoPlayback)Ref(loop,"echoPrefab");var copy=(GameObject)PrefabUtility.InstantiatePrefab(original.gameObject);copy.name="EchoReconstructedAdult";var body=copy.GetComponentsInChildren<MeshFilter>(true).Single(f=>f.name=="EP_AdultRecordedSurface");body.sharedMesh=mesh;body.GetComponent<Renderer>().sharedMaterials=recorded;var depth=body.transform.Find("Recorded self depth").GetComponent<MeshFilter>();depth.sharedMesh=mesh;var dm=depth.GetComponent<Renderer>().sharedMaterial;depth.GetComponent<Renderer>().sharedMaterials=slots.Select(_=>dm).ToArray();var saved=PrefabUtility.SaveAsPrefabAsset(copy,"Assets/_Game/Prefabs/Reconstruction/EchoReconstructedAdult.prefab");Object.DestroyImmediate(copy);Set(loop,"echoPrefab",saved.GetComponent<EchoPlayback>());
var review=Object.FindFirstObjectByType<FinalDoorMemory>();Set(review,"directWitnessMaterial",recorded[Array.IndexOf(slots,"FIG_Skin")]);var passage=Object.FindFirstObjectByType<VS1ParallelFigureEvent>();var figure=(GameObject)Ref(passage,"figureRoot");var visible=figure.GetComponentsInChildren<MeshFilter>(true).Single(f=>f.name=="SE2 physical adult / same authored silhouette");visible.sharedMesh=mesh;visible.GetComponent<Renderer>().sharedMaterials=physical;
var source=Object.FindObjectsByType<AudioSource>(FindObjectsInactive.Include,FindObjectsSortMode.None).FirstOrDefault(a=>a.name=="6B review latch");foreach(var terminal in Object.FindObjectsByType<TemporalRelayTerminal>(FindObjectsInactive.Include,FindObjectsSortMode.None)){var a=terminal.GetComponent<AudioSource>();if(!a)a=terminal.gameObject.AddComponent<AudioSource>();a.clip=AssetDatabase.LoadAssetAtPath<AudioClip>("Assets/_Game/Audio/Clips/Finish/Relay.wav");a.playOnAwake=false;a.loop=false;a.spatialBlend=1;a.volume=.13f;a.minDistance=1;a.maxDistance=5;a.dopplerLevel=0;if(source)a.outputAudioMixerGroup=source.outputAudioMixerGroup;Set(terminal,"contact",a);}
Physics.SyncTransforms();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);AssetDatabase.SaveAssets();var report=new{mesh=path,vertices=mesh.vertexCount,submeshes=mesh.subMeshCount,slots,prefab=AssetDatabase.GetAssetPath(saved),bodyScale=new[]{body?0f:1f},rigs=saved.GetComponentsInChildren<Animator>(true).Length,audio=saved.GetComponentsInChildren<AudioSource>(true).Length};File.WriteAllText("Docs/VisualEvidence/2026-10-02-Reconstruction/model-authoring.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));return report;
}}
