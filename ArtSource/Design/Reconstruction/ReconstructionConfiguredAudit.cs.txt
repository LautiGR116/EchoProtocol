using System;using System.IO;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using UnityEngine.SceneManagement;using EchoProtocol.Puzzle;using EchoProtocol.Echo;using EchoProtocol.Campaign;using EchoProtocol.Presentation;using EchoProtocol.Interaction;using Object=UnityEngine.Object;
public static class ReconstructionConfiguredAudit {
static Object Ref(Object o,string n)=>new SerializedObject(o).FindProperty(n).objectReferenceValue;
static T F<T>()where T:Component=>Object.FindObjectsByType<T>(FindObjectsInactive.Include,FindObjectsSortMode.None).OrderByDescending(t=>t.gameObject.activeInHierarchy).First();
static void Need(bool ok,string s){if(!ok)throw new Exception(s);}
static float[] V(Vector3 v)=>new[]{v.x,v.y,v.z};
public static object Main(){var scene=SceneManager.GetActiveScene();Need(!EditorApplication.isPlaying&&!EditorApplication.isCompiling&&!EditorApplication.isUpdating&&!scene.isDirty,"Saved idle Edit scene required");Need(scene.path=="Assets/_Game/Scenes/Production/EchoProtocol_Reconstruction.unity","Wrong scene");int missing=scene.GetRootGameObjects().Sum(g=>g.GetComponentsInChildren<Transform>(true).Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)));Need(missing==0,"Missing script");var p1=F<ParallelObservationRelay>();var p3=F<RecombinationRelay>();Need(p1.RequiresSimultaneousPresence&&p3.RequiresPairedTransfer,"Reconstruction puzzle modes not enabled");Need(Mathf.Approximately(new SerializedObject(p1).FindProperty("cooperationSeconds").floatValue,.65f),"Wrong coincidence duration");var hints=F<ContextualHints>();var hs=new SerializedObject(hints);Need(hs.FindProperty("firstDelay").floatValue==45&&hs.FindProperty("nextDelay").floatValue==65,"Test clock leaked into authored hints");Need(Ref(hints,"session")==F<GameSession>()&&Ref(hints,"guidance")==F<ProtocolGuidance>()&&Ref(hints,"loop")==F<LoopController>(),"Hint refs mismatch");Need(Ref(F<GameSession>(),"contextualHints")==hints,"Session lacks optional hints");var terminals=Object.FindObjectsByType<TemporalRelayTerminal>(FindObjectsInactive.Include,FindObjectsSortMode.None);Need(terminals.Length==2&&terminals.Count(t=>t.FinalReceiver)==1,"Two distinct receivers required");foreach(var t in terminals){Need(Ref(t,"relay")==p3&&Ref(t,"pilot")&&Ref(t,"status")&&Ref(t,"contact"),"Receiver reference missing");Need(!t.GetComponent<BoxCollider>().isTrigger,"Receiver physical focus missing");}
var loop=F<LoopController>();var prefab=(EchoPlayback)Ref(loop,"echoPrefab");Need(AssetDatabase.GetAssetPath(prefab)=="Assets/_Game/Prefabs/Reconstruction/EchoReconstructedAdult.prefab","Wrong Echo asset");var body=prefab.GetComponentsInChildren<MeshFilter>(true).Single(f=>f.name=="EP_AdultRecordedSurface");var depth=body.transform.Find("Recorded self depth").GetComponent<MeshFilter>();Need(body.sharedMesh==depth.sharedMesh,"Self-depth mesh mismatch");var mats=body.GetComponent<Renderer>().sharedMaterials;Need(mats.Length==10&&mats.All(m=>m&&m.shader.name=="EchoProtocol/RecordedWitness"),"Wrong display palette");Need(prefab.GetComponentsInChildren<Animator>(true).Length==0&&prefab.GetComponentsInChildren<AudioSource>(true).Length==0,"Unrecorded runtime channel");var messages=ShaderUtil.GetShaderMessages(mats[0].shader);Need(!messages.Any(m=>m.severity.ToString()=="Error"),"Shader error");Need(Ref(F<FinalDoorMemory>(),"directWitnessMaterial")==mats.Single(m=>m.name=="M_Record_FIG_Skin"),"Final witness surface mismatch");Need(new SerializedObject(F<CampaignDirector>()).FindProperty("checkpointKey").stringValue=="EP.CampaignCheckpoint.Reconstruction.v1","Wrong save namespace");Need(EditorBuildSettings.scenes.Count(s=>s.enabled)==1&&EditorBuildSettings.scenes.Single(s=>s.enabled).path==scene.path,"Build scene mismatch");
var hud=F<InteractionFocusHUD>();Need(Ref(hud,"interactor")==F<PlayerInteractor>()&&Ref(hud,"session")==F<GameSession>(),"HUD refs mismatch");var report=new{status="PASS",scene=scene.path,missingScripts=missing,mesh=AssetDatabase.GetAssetPath(body.sharedMesh),vertices=body.sharedMesh.vertexCount,triangles=body.sharedMesh.triangles.Length/3,meshBounds=V(body.sharedMesh.bounds.size),bodyLocalScale=V(body.transform.localScale),materials=mats.Select(m=>m.name).ToArray(),recordedChannels="Root position/body yaw only; no Animator or AudioSource",p1CooperationSeconds=.65f,p3Receivers=terminals.Select(t=>new{t.name,t.FinalReceiver,position=V(t.transform.position)}).ToArray(),firstHintSeconds=45,nextHintSeconds=65,shaderSupported=mats[0].shader.isSupported,shaderMessages=messages.Select(m=>m.message).ToArray(),saveKey="EP.CampaignCheckpoint.Reconstruction.v1",buildScenes=EditorBuildSettings.scenes.Where(s=>s.enabled).Select(s=>s.path).ToArray()};File.WriteAllText("Docs/VisualEvidence/2026-10-02-Reconstruction/configured-audit.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));return report;
}}
