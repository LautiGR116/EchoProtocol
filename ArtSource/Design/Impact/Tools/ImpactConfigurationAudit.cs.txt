using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using EchoProtocol.Campaign;using EchoProtocol.Audio;using EchoProtocol.Echo;using EchoProtocol.Player;using EchoProtocol.Presentation;
public static class ImpactConfigurationAudit{
 static T F<T>(UnityEngine.Object o,string n)where T:UnityEngine.Object=>(T)new SerializedObject(o).FindProperty(n).objectReferenceValue;
 public static object Main(){
  var c=UnityEngine.Object.FindFirstObjectByType<CampaignDirector>();var p=UnityEngine.Object.FindFirstObjectByType<PlayerEchoClearance>();var t=UnityEngine.Object.FindFirstObjectByType<MachineCoverThreat>();var a=UnityEngine.Object.FindFirstObjectByType<MachineCoverPresence>();var mix=UnityEngine.Object.FindFirstObjectByType<SessionAudio>();
  if(!c||!p||!t||!a||!mix||F<LoopController>(p,"loop")!=F<LoopController>(c,"loop")||!F<LoopController>(p,"loop").gameObject.activeInHierarchy||F<MachineCoverPresence>(t,"presence")!=a||F<MachineCoverThreat>(a,"threat")!=t)throw new Exception("Active campaign/presence reference mismatch");
  var so=new SerializedObject(a);string[] references={"encounter","threat","session","listener","frameContact","warning","rush","friction","score","threatPractical"};foreach(string n in references)if(!so.FindProperty(n).objectReferenceValue)throw new Exception("Missing "+n);
  foreach(string n in new[]{"spatialSources","boundaries","steps"}){var q=so.FindProperty(n);if(q.arraySize<2)throw new Exception(n);for(int i=0;i<q.arraySize;i++)if(!q.GetArrayElementAtIndex(i).objectReferenceValue)throw new Exception("Null "+n);}
  var paths=AssetDatabase.FindAssets("t:AudioClip",new[]{"Assets/_Game/Audio/Clips/Impact"}).Select(AssetDatabase.GUIDToAssetPath).ToArray();if(paths.Length!=7)throw new Exception("Seven traceable masters required");
  var import=paths.Select(path=>{var ai=(AudioImporter)AssetImporter.GetAtPath(path);var clip=AssetDatabase.LoadAssetAtPath<AudioClip>(path);return new{path,clip.channels,clip.frequency,clip.length,ai.forceToMono,loadType=ai.defaultSampleSettings.loadType.ToString(),compression=ai.defaultSampleSettings.compressionFormat.ToString(),normalize=ai.GetType().GetProperty("normalize")?.GetValue(ai)};}).ToArray();
  var tr=new SerializedObject(t);var result=new{status="PASS",scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,activeLoop=F<LoopController>(p,"loop").name,referenceFields=references.Length,spatialSources=so.FindProperty("spatialSources").arraySize,steps=so.FindProperty("steps").arraySize,normalCeiling=new SerializedObject(mix).FindProperty("normalCeiling").floatValue,pursuitSpeed=tr.FindProperty("pursuitSpeed").floatValue,captureDistance=tr.FindProperty("captureDistance").floatValue,navigationClearance=tr.FindProperty("navigationClearance").floatValue,colliderRadius=t.GetComponent<CharacterController>().radius,assets=import};Directory.CreateDirectory("Docs/VisualEvidence/2026-10-11-Impact");File.WriteAllText("Docs/VisualEvidence/2026-10-11-Impact/configuration.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));return "PASS active references, seven masters and authored collision/navigation settings";
 }
}
