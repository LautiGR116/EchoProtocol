using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class StyleFolioSigns{
static float Width(Font f,string text,int size,FontStyle style){f.RequestCharactersInTexture(text,size,style);float maximum=0;foreach(var line in text.Split('\n')){float width=0;foreach(char c in line){if(f.GetCharacterInfo(c,out var info,size,style))width+=info.advance;}maximum=Mathf.Max(maximum,width);}return maximum;}
public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Edit only");var f=AssetDatabase.LoadAssetAtPath<Font>("Assets/_Game/Art/Folio/Fonts/IBMPlexSans-Medium.ttf");var labels=UnityEngine.Object.FindObjectsByType<TextMesh>(FindObjectsSortMode.None);var changes=labels.Select(t=>{float old=Width(t.font,t.text,t.fontSize,t.fontStyle),next=Width(f,t.text,t.fontSize,FontStyle.Normal);float ratio=next>0?Mathf.Clamp(old/next,.6f,1.2f):1;t.font=f;t.fontStyle=FontStyle.Normal;t.characterSize*=ratio;t.lineSpacing=1.12f;var renderer=t.GetComponent<Renderer>();if(renderer){var m=new Material(renderer.sharedMaterial);m.mainTexture=f.material.mainTexture;m.name="Folio / physical type "+t.name;renderer.sharedMaterial=m;}return new{path=t.name,t.text,ratio,t.characterSize};}).ToArray();EditorSceneManager.MarkSceneDirty(UnityEngine.SceneManagement.SceneManager.GetActiveScene());EditorSceneManager.SaveOpenScenes();File.WriteAllText("Docs/VisualEvidence/2026-10-08-Folio/signage-integration.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{scope="Active physical lettering only; measured font advance keeps inherited authored width. Same transforms, panels, collision and state owners.",labels=changes},Newtonsoft.Json.Formatting.Indented));return labels.Length+" physical text surfaces unified";}}
