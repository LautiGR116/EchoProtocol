using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using EchoProtocol.Presentation;
public static class PersistFolioType{public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Edit only");string dir="Assets/_Game/Art/Folio/Materials";Directory.CreateDirectory(dir);var labels=UnityEngine.Object.FindObjectsByType<TextMesh>(FindObjectsSortMode.None);int i=0,added=0;foreach(var text in labels){var r=text.GetComponent<Renderer>();var mat=new Material(r.sharedMaterial);mat.name="Folio physical type "+text.name;AssetDatabase.CreateAsset(mat,dir+"/Sign-"+(i++).ToString("00")+".mat");r.sharedMaterial=mat;var type=text.GetComponent<DiegeticText>();if(!type){type=text.gameObject.AddComponent<DiegeticText>();var s=new SerializedObject(type);s.FindProperty("depthTestedShader").objectReferenceValue=mat.shader.name=="GUI/Text Shader"?AssetDatabase.LoadAssetAtPath<Shader>("Assets/_Game/Shaders/DiegeticText.shader"):mat.shader;s.ApplyModifiedPropertiesWithoutUndo();added++;}}AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(UnityEngine.SceneManagement.SceneManager.GetActiveScene());EditorSceneManager.SaveOpenScenes();return "Persistent lettering materials="+i+" added atlas owners="+added;}}
