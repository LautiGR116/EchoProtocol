 static string Build(bool qa){if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");
 var path="Builds/EchoProtocol-Folio-"+(qa?"QA-":"")+"20261008/EchoProtocol.app";
 var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/_Game/Scenes/Production/EchoProtocol_Folio.unity"},locationPathName=path,target=BuildTarget.StandaloneOSX,options=BuildOptions.DetailedBuildReport,extraScriptingDefines=qa?new[]{"ECHO_NATIVE_VERIFY"}:new string[0]});
