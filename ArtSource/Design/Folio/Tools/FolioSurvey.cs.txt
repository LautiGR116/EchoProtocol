using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class FolioSurvey{
static string PathOf(Transform t){return t.parent?PathOf(t.parent)+"/"+t.name:t.name;}
public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Stop before survey");var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(scene.path!="Assets/_Game/Scenes/Production/EchoProtocol_Resonance.unity")throw new Exception(scene.path);var labels=UnityEngine.Object.FindObjectsByType<TextMesh>(FindObjectsInactive.Include,FindObjectsSortMode.None);var scripts=UnityEngine.Object.FindObjectsByType<MonoBehaviour>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(x=>x&&x.GetType().GetMethod("OnGUI",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance)!=null).Select(x=>new{path=PathOf(x.transform),type=x.GetType().FullName,x.enabled,active=x.gameObject.activeInHierarchy}).ToArray();var docs=UnityEngine.Object.FindObjectsByType<EchoProtocol.Campaign.HouseInteraction>(FindObjectsInactive.Include,FindObjectsSortMode.None).Select(x=>{var s=new SerializedObject(x);return new{path=PathOf(x.transform),title=s.FindProperty("documentTitle").stringValue,body=s.FindProperty("documentBody").stringValue,active=x.gameObject.activeInHierarchy};}).ToArray();var report=new{scene=scene.path,sceneDirty=scene.isDirty,labels=labels.Select(t=>new{path=PathOf(t.transform),t.text,t.fontSize,t.characterSize,t.lineSpacing,font=t.font?t.font.name:null,active=t.gameObject.activeInHierarchy,renderer=t.GetComponent<Renderer>().enabled,position=t.transform.position,scale=t.transform.localScale,bounds=t.GetComponent<Renderer>().bounds}).ToArray(),scripts,documents=docs};File.WriteAllText("Docs/VisualEvidence/2026-10-08-Folio/baseline-text-survey.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented,new Newtonsoft.Json.JsonSerializerSettings{ReferenceLoopHandling=Newtonsoft.Json.ReferenceLoopHandling.Ignore}));return "TextMeshes="+labels.Length+" OnGUI="+scripts.Length+" documents="+docs.Length+" clean="+!scene.isDirty;}}
