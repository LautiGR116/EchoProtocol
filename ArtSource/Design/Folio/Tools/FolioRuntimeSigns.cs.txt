using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using EchoProtocol.Presentation;
public static class FolioRuntimeSigns{
static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
static float Width(Font f,string text,int size,FontStyle style){f.RequestCharactersInTexture(text,size,style);float maximum=0;foreach(var line in text.Split('\n')){float width=0;foreach(char c in line){if(f.GetCharacterInfo(c,out var info,size,style))width+=info.advance;}maximum=Mathf.Max(maximum,width);}return maximum;}
public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");var f=AssetDatabase.LoadAssetAtPath<Font>("Assets/_Game/Art/Folio/Fonts/IBMPlexSans-Medium.ttf");var labels=UnityEngine.Object.FindObjectsByType<TextMesh>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(t=>t.name=="Carried reserve identity"||PathOf(t.transform).Contains("Return / illuminated subsuelo direction/")||t.name=="Reserve unit identity"&&t.font!=f).ToArray();if(labels.Length!=5)throw new Exception("Expected five runtime-activated labels, got "+labels.Length);
var changes=labels.Select((t,i)=>{if(!t.GetComponent<DiegeticText>())throw new Exception("Expected existing atlas owner: "+PathOf(t.transform));float old=Width(t.font,t.text,t.fontSize,t.fontStyle),next=Width(f,t.text,t.fontSize,FontStyle.Normal);float ratio=next>0?Mathf.Clamp(old/next,.6f,1.2f):1;t.font=f;t.fontStyle=FontStyle.Normal;t.characterSize*=ratio;t.lineSpacing=1.12f;var r=t.GetComponent<Renderer>();var mat=new Material(r.sharedMaterial);mat.mainTexture=f.material.mainTexture;mat.name="Folio runtime sign "+t.name;AssetDatabase.CreateAsset(mat,"Assets/_Game/Art/Folio/Materials/Sign-"+(51+i).ToString("00")+".mat");r.sharedMaterial=mat;return new{path=PathOf(t.transform),t.text,ratio,t.characterSize};}).ToArray();AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(UnityEngine.SceneManagement.SceneManager.GetActiveScene());EditorSceneManager.SaveOpenScenes();File.WriteAllText("Docs/VisualEvidence/2026-10-08-Folio/runtime-signage.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{scope="Five initially hidden signs enabled by actual reserve/return progression. Existing atlas owners; same supports, positions and state references.",labels=changes},Newtonsoft.Json.Formatting.Indented));return "Five runtime signs unified";}}
