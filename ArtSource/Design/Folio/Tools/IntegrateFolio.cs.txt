using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using EchoProtocol.Presentation;
public static class IntegrateFolio{
public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Edit only");if(UnityEngine.SceneManagement.SceneManager.GetActiveScene().name!="EchoProtocol_Folio")throw new Exception("Wrong scene");AssetDatabase.Refresh();string root="Assets/_Game/Art/Folio/";Directory.CreateDirectory(root+"Textures");foreach(bool domestic in new[]{true,false}){var texture=FolioUI.MakePaper(domestic);File.WriteAllBytes(root+"Textures/"+(domestic?"PersonalPaper":"ArchivePaper")+".png",texture.EncodeToPNG());UnityEngine.Object.DestroyImmediate(texture);}AssetDatabase.Refresh();
foreach(var guid in AssetDatabase.FindAssets("t:Font",new[]{root+"Fonts"})){var path=AssetDatabase.GUIDToAssetPath(guid);var importer=(TrueTypeFontImporter)AssetImporter.GetAtPath(path);importer.fontTextureCase=FontTextureCase.Dynamic;importer.includeFontData=true;importer.fontNames=new[]{AssetDatabase.LoadAssetAtPath<Font>(path).fontNames[0]};importer.SaveAndReimport();}
foreach(var guid in AssetDatabase.FindAssets("t:Texture2D",new[]{root+"Textures"})){var p=AssetDatabase.GUIDToAssetPath(guid);var i=(TextureImporter)AssetImporter.GetAtPath(p);i.textureType=TextureImporterType.Default;i.mipmapEnabled=false;i.sRGBTexture=true;i.alphaSource=TextureImporterAlphaSource.None;i.textureCompression=TextureImporterCompression.Uncompressed;i.maxTextureSize=512;i.SaveAndReimport();}
var g=new GameObject("Folio / shared graphic identity");var ui=g.AddComponent<FolioUI>();var so=new SerializedObject(ui);string[] properties={"displayFont","narrativeItalic","functionalFont","functionalMedium","editorialFont","editorialItalic"};string[] files={"CormorantGaramond-SemiBold.ttf","CormorantGaramond-Italic.ttf","IBMPlexSans-Regular.ttf","IBMPlexSans-Medium.ttf","IBMPlexSerif-Regular.ttf","IBMPlexSerif-Italic.ttf"};for(int i=0;i<files.Length;i++)so.FindProperty(properties[i]).objectReferenceValue=AssetDatabase.LoadAssetAtPath<Font>(root+"Fonts/"+files[i]);so.FindProperty("domesticPaper").objectReferenceValue=AssetDatabase.LoadAssetAtPath<Texture2D>(root+"Textures/PersonalPaper.png");so.FindProperty("archivePaper").objectReferenceValue=AssetDatabase.LoadAssetAtPath<Texture2D>(root+"Textures/ArchivePaper.png");so.FindProperty("subtitles").objectReferenceValue=UnityEngine.Object.FindFirstObjectByType<EchoProtocol.Horror.SD1SubtitleDisplay>();so.ApplyModifiedPropertiesWithoutUndo();var session=UnityEngine.Object.FindFirstObjectByType<GameSession>();var ss=new SerializedObject(session);ss.FindProperty("identity").objectReferenceValue=ui;ss.FindProperty("editionLabel").stringValue="FOLIO / ARCHIVO 6B";ss.ApplyModifiedPropertiesWithoutUndo();EditorSceneManager.MarkSceneDirty(UnityEngine.SceneManagement.SceneManager.GetActiveScene());EditorSceneManager.SaveOpenScenes();return "Folio fonts/paper/session configured";}}
