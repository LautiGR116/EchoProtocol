using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;
public static class FolioSignViews{
static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
public static string Main(){if(!EditorApplication.isPlaying)throw new Exception("Live font atlas required");var labels=UnityEngine.Object.FindObjectsByType<TextMesh>(FindObjectsInactive.Include,FindObjectsSortMode.None).Where(t=>t.name=="Carried reserve identity"||PathOf(t.transform).Contains("Return / illuminated subsuelo direction/")||t.name=="Reserve unit identity"&&t.transform.parent.name!="Reserve unit / RECIBIDOR").ToArray();if(labels.Length!=5)throw new Exception("Expected five signs");string folder="Docs/VisualEvidence/2026-10-08-Folio/runtime-sign-views";Directory.CreateDirectory(folder);var original=Camera.main;var go=new GameObject("Temporary explicit font inspection camera");var camera=go.AddComponent<Camera>();camera.CopyFrom(original);camera.enabled=false;camera.fieldOfView=38;var rt=new RenderTexture(1024,512,24);rt.Create();camera.targetTexture=rt;int i=0;foreach(var text in labels){for(var t=text.transform;t;t=t.parent)t.gameObject.SetActive(true);text.font.RequestCharactersInTexture(text.text,text.fontSize,text.fontStyle);var atlas=text.font.material.mainTexture;var renderer=text.GetComponent<Renderer>();renderer.material.mainTexture=atlas;var bounds=renderer.bounds;float distance=Mathf.Max(.15f,bounds.size.magnitude*1.35f);for(int side=0;side<2;side++){camera.transform.position=bounds.center+text.transform.forward*distance*(side==0?1:-1);camera.transform.LookAt(bounds.center,text.transform.up);camera.Render();var previous=RenderTexture.active;RenderTexture.active=rt;var image=new Texture2D(1024,512,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,1024,512),0,0);image.Apply();RenderTexture.active=previous;File.WriteAllBytes(folder+"/sign-"+i+"-side-"+side+".png",image.EncodeToPNG());UnityEngine.Object.Destroy(image);}i++;}camera.targetTexture=null;rt.Release();UnityEngine.Object.Destroy(rt);UnityEngine.Object.Destroy(go);File.WriteAllText(folder+"/scope.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{scope="Explicit close inspection cameras and activation of five initially hidden signs in live Play mode. Same production font, material shader, text, placement and lights; not normal Player approach or campaign completion. No scene save.",labels=labels.Select(t=>new{path=PathOf(t.transform),t.text,font=t.font.name}).ToArray()},Newtonsoft.Json.Formatting.Indented));return "Five live labels rendered from both sides";}}
