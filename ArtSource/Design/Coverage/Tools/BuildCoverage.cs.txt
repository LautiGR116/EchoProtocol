using System;using System.IO;using UnityEditor;using UnityEditor.Build.Reporting;
public static class BuildCoverage{
public static string QA()=>Build(true);public static string Delivery()=>Build(false);
static string Build(bool qa){
if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");
string path="Builds/EchoProtocol-Coverage-"+(qa?"QA-":"")+"20261009/EchoProtocol.app";
var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/_Game/Scenes/Production/EchoProtocol_Coverage.unity"},locationPathName=path,target=BuildTarget.StandaloneOSX,options=BuildOptions.DetailedBuildReport,extraScriptingDefines=qa?new[]{"ECHO_NATIVE_VERIFY"}:new string[0]});
string output="Docs/VisualEvidence/2026-10-09-Coverage";Directory.CreateDirectory(output);
File.WriteAllText(output+"/"+(qa?"qa-build":"delivery-build")+".json",Newtonsoft.Json.JsonConvert.SerializeObject(new{result=report.summary.result.ToString(),errors=report.summary.totalErrors,warnings=report.summary.totalWarnings,bytes=report.summary.totalSize,seconds=report.summary.totalTime.TotalSeconds,output=report.summary.outputPath,qa,development=false},Newtonsoft.Json.Formatting.Indented));
return report.summary.result.ToString();}}
