static void R(UnityEngine.Object o,string f,UnityEngine.Object v){var so=new SerializedObject(o);so.FindProperty(f).objectReferenceValue=v;so.ApplyModifiedPropertiesWithoutUndo();}
static GameObject Shape(string n,PrimitiveType type,Vector3 p,Vector3 scale,Material mat,Transform parent=null){var g=GameObject.CreatePrimitive(type);g.name=n;g.transform.SetParent(parent?parent:root);g.transform.position=p;g.transform.localScale=scale;g.GetComponent<Renderer>().sharedMaterial=mat;UnityEngine.Object.DestroyImmediate(g.GetComponent<Collider>());return g;}
static Mesh Ring(){
var m=new Mesh{name="Coverage machined handwheel ring"};var v=new Vector3[48*10];var uv=new Vector2[v.Length];var tr=new int[v.Length*6];int k=0;
for(int i=0;i<48;i++)for(int j=0;j<10;j++){float a=i*Mathf.PI*2/48,b=j*Mathf.PI*2/10,r=.175f+.018f*Mathf.Cos(b);v[i*10+j]=new Vector3(r*Mathf.Cos(a),r*Mathf.Sin(a),.018f*Mathf.Sin(b));uv[i*10+j]=new Vector2(i/48f,j/10f);int c=i*10+j,d=((i+1)%48)*10+j,e=((i+1)%48)*10+(j+1)%10,f=i*10+(j+1)%10;tr[k++]=c;tr[k++]=d;tr[k++]=e;tr[k++]=c;tr[k++]=e;tr[k++]=f;}
m.vertices=v;m.uv=uv;m.triangles=tr;m.RecalculateNormals();m.RecalculateBounds();AssetDatabase.CreateAsset(m,"Assets/_Game/Art/Coverage/Handwheel.asset");return m;}
