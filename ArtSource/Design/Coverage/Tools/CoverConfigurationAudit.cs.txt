using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;using EchoProtocol.Campaign;using EchoProtocol.Echo;using EchoProtocol.Audio;
public static class CoverConfigurationAudit {
public static string Main(){
if(EditorApplication.isPlaying)throw new Exception("Stopped scene audit required");var cover=UnityEngine.Object.FindFirstObjectByType<MachineCoverEncounter>();var objects=new UnityEngine.Object[]{cover,UnityEngine.Object.FindFirstObjectByType<RecordedMachineControl>(),UnityEngine.Object.FindFirstObjectByType<MachineCoverThreat>()}.Concat(UnityEngine.Object.FindObjectsByType<MachineCoverInteraction>(FindObjectsSortMode.None)).ToArray();int checkedRefs=0;
foreach(var o in objects){if(!o)throw new Exception("Missing encounter object");var s=new SerializedObject(o);var p=s.GetIterator();while(p.NextVisible(true))if(p.propertyType==SerializedPropertyType.ObjectReference&&p.name!="m_Script"){if(!p.objectReferenceValue)throw new Exception(o.name+" missing "+p.propertyPath);checkedRefs++;}}
var gates=(new SerializedObject(cover)).FindProperty("gateAudio");if(gates.arraySize!=3)throw new Exception("Gate audio contract");
foreach(var a in UnityEngine.Object.FindObjectsByType<SlidingDoorAudio>(FindObjectsSortMode.None)){var s=new SerializedObject(a);if(s.FindProperty("door").objectReferenceValue&&s.FindProperty("mechanismSource").objectReferenceValue==null)throw new Exception("Missing routed door source");}
var loop=UnityEngine.Object.FindFirstObjectByType<LoopController>();var l=new SerializedObject(loop);var controls=l.FindProperty("machineControls");if(controls.arraySize!=1)throw new Exception("One bounded recorded target required");if(UnityEngine.Object.FindFirstObjectByType<PressureIsolation>().enabled)throw new Exception("Legacy isolation owner still active");
var key=(new SerializedObject(UnityEngine.Object.FindFirstObjectByType<CampaignDirector>())).FindProperty("checkpointKey").stringValue;
if(key!="EP.CampaignCheckpoint.Coverage.v1")throw new Exception("Checkpoint namespace");
var sources=cover.GetComponentsInChildren<AudioSource>(true).Select(s=>new{name=s.name,clip=s.clip?AssetDatabase.GetAssetPath(s.clip):"",spatial=s.spatialBlend,volume=s.volume,routed=s.outputAudioMixerGroup?s.outputAudioMixerGroup.name:"",playOnAwake=s.playOnAwake}).ToArray();
var result=new{status="PASS",explicitObjectRefs=checkedRefs,recordedControls=controls.arraySize,checkpointKey=key,sources,scope="Actual configured candidate scene and explicit references. Perceptual credibility remains human playtest work."};File.WriteAllText("Docs/VisualEvidence/2026-10-09-Coverage/configuration.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));return "PASS "+checkedRefs+" scene references";
}}
