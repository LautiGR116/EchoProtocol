using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;
public static class ResonanceVisualAudit{
 static string P(Transform t)=>t.parent?P(t.parent)+"/"+t.name:t.name;
 static string Vec(Vector3 v)=>v.x+","+v.y+","+v.z;
 static Dictionary<string,string> Visual(Scene s){var d=new Dictionary<string,string>();foreach(var root in s.GetRootGameObjects()){
 foreach(var l in root.GetComponentsInChildren<Light>(true))d["Light:"+P(l.transform)]=Vec(l.transform.position)+"/"+Vec(l.transform.eulerAngles)+"/"+l.enabled+"/"+l.gameObject.activeSelf+"/"+l.color+"/"+l.intensity+"/"+l.range+"/"+l.spotAngle+"/"+l.type;
 foreach(var c in root.GetComponentsInChildren<Collider>(true)){if(P(c.transform).StartsWith("Resonance /"))continue;d["Collider:"+P(c.transform)]=Vec(c.transform.position)+"/"+Vec(c.transform.eulerAngles)+"/"+Vec(c.transform.lossyScale)+"/"+c.enabled+"/"+c.isTrigger+"/"+(c is BoxCollider b?Vec(b.center)+"/"+Vec(b.size):c.GetType().Name);}
 foreach(var r in root.GetComponentsInChildren<Renderer>(true))d["Renderer:"+P(r.transform)]=Vec(r.transform.position)+"/"+Vec(r.transform.eulerAngles)+"/"+Vec(r.transform.lossyScale)+"/"+r.enabled+"/"+r.gameObject.activeSelf+"/"+string.Join(",",r.sharedMaterials.Select(m=>AssetDatabase.GetAssetPath(m)));}
 foreach(var root in s.GetRootGameObjects())foreach(var b in root.GetComponentsInChildren<EchoProtocol.Presentation.LightShadowBudget>(true)){var so=new SerializedObject(b);string RefNames(string name){var a=so.FindProperty(name);return string.Join(";",Enumerable.Range(0,a.arraySize).Select(i=>a.GetArrayElementAtIndex(i).objectReferenceValue is Component c?P(c.transform):"null"));}d["Budget:"+P(b.transform)]=RefNames("lights")+"/"+RefNames("residenceLights")+"/"+so.FindProperty("incrementalShadowChanges").boolValue;}
 return d;}
 public static string Main(){if(EditorApplication.isPlaying)throw new Exception("Stopped Editor required");var current=SceneManager.GetActiveScene();var candidate=Visual(current);var old=EditorSceneManager.OpenScene("Assets/_Game/Scenes/Production/EchoProtocol_Lighting.unity",OpenSceneMode.Additive);
 try{var baseline=Visual(old);var diffs=baseline.Where(x=>!candidate.TryGetValue(x.Key,out var v)||v!=x.Value).Select(x=>x.Key).ToArray();if(diffs.Length>0){File.WriteAllText("/tmp/resonance-visual-diff.json",Newtonsoft.Json.JsonConvert.SerializeObject(diffs.Select(k=>new{key=k,baseline=baseline[k],candidate=candidate.ContainsKey(k)?candidate[k]:"MISSING"}),Newtonsoft.Json.Formatting.Indented));throw new Exception("Visual/physics changes: "+string.Join(";",diffs.Take(15)));}var sources=current.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<EchoProtocol.Audio.ResonanceEmitter>(true)).ToArray();var clips=AssetDatabase.FindAssets("t:AudioClip",new[]{"Assets/_Game/Audio/Clips/Resonance"}).Select(g=>AssetDatabase.LoadAssetAtPath<AudioClip>(AssetDatabase.GUIDToAssetPath(g))).ToArray();if(sources.Length!=8||clips.Length!=13||clips.Any(c=>c.channels!=1))throw new Exception("Sound inventory mismatch");
 File.WriteAllText("Docs/AudioEvidence/2026-10-08-Resonance/scene-audit.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{status="PASS",scope="Existing transforms, light intensity/color/range, enabled state, render materials, colliders and shadow-budget memberships compared. Instantaneous Light.shadows is camera-owned and excluded from serialized design equality.",baselineObjects=baseline.Count,lights=baseline.Keys.Count(k=>k.StartsWith("Light:")),colliders=baseline.Keys.Count(k=>k.StartsWith("Collider:")),renderers=baseline.Keys.Count(k=>k.StartsWith("Renderer:")),changedExistingVisualOrPhysicsObjects=diffs,newTriggerOnly=true,sources=sources.Select(e=>new{name=e.name,clip=e.Source.clip.name,channels=e.Source.clip.channels,group=e.Source.outputAudioMixerGroup.name,blend=e.Source.spatialBlend,loop=e.Source.loop}).ToArray(),clips=clips.Select(c=>new{name=c.name,c.length,c.channels,c.frequency,load=c.loadType.ToString()}).ToArray()},Newtonsoft.Json.Formatting.Indented));return "PASS visual and physics preserved; 8 routed sources and 13 mono masters";}
 finally{EditorSceneManager.CloseScene(old,true);SceneManager.SetActiveScene(current);}}
}
