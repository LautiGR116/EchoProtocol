using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;using UnityEngine.Audio;
public static class AudioInventory{
public static string Main(){
var ss=UnityEngine.Object.FindObjectsByType<AudioSource>(FindObjectsInactive.Include,FindObjectsSortMode.None);AudioSettings.GetDSPBufferSize(out int buffer,out int buffers);
var mix=AssetDatabase.FindAssets("t:AudioMixer",new[]{"Assets"}).Select(g=>{var p=AssetDatabase.GUIDToAssetPath(g);var m=AssetDatabase.LoadAssetAtPath<AudioMixer>(p);return new{path=p,groups=m.FindMatchingGroups("").Select(x=>x.name).ToArray(),master=m.GetFloat("MasterVolume",out float v)?(float?)v:null};}).ToArray();
var report=new{scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,sampleRate=AudioSettings.outputSampleRate,buffer,buffers,configuration=AudioSettings.GetConfiguration(),listeners=UnityEngine.Object.FindObjectsByType<AudioListener>(FindObjectsInactive.Include,FindObjectsSortMode.None).Select(l=>new{path=PathOf(l.transform),l.enabled}).ToArray(),mixers=mix,sources=ss.Select(s=>new{path=PathOf(s.transform),pos=s.transform.position,clip=s.clip?AssetDatabase.GetAssetPath(s.clip):null,seconds=s.clip?s.clip.length:0,channels=s.clip?s.clip.channels:0,load=s.clip?s.clip.loadType.ToString():null,s.volume,s.pitch,s.spatialBlend,s.minDistance,s.maxDistance,rolloff=s.rolloffMode.ToString(),s.playOnAwake,s.loop,s.dopplerLevel,group=s.outputAudioMixerGroup?s.outputAudioMixerGroup.name:null,components=s.GetComponents<Component>().Select(x=>x?x.GetType().Name:"MISSING").ToArray()}).ToArray()};
File.WriteAllText("Docs/AudioEvidence/2026-10-08-Resonance/final-inventory.json",Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented,new Newtonsoft.Json.JsonSerializerSettings{ReferenceLoopHandling=Newtonsoft.Json.ReferenceLoopHandling.Ignore}));
return "sources="+ss.Length+" listeners="+report.listeners.Length+" rate="+report.sampleRate+" buffer="+buffer+"x"+buffers+" mixers="+Newtonsoft.Json.JsonConvert.SerializeObject(mix);}
static string PathOf(Transform t){return t.parent?PathOf(t.parent)+"/"+t.name:t.name;}}
